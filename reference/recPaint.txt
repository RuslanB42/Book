function recPaint(box) {
  if (!REC.st.onboarded && recPositive() < 3) { recStart(box); return; }
  REC.queue = recRank();
  const head = h('div', { class: 'fy-head' }, h('p', { class: 'fy-count' }, recCountLabel()),
    h('button', { class: 'icon-btn fy-tune', 'aria-label': 'Настроить подборку', onclick: () => recTune(box) }, icon('sliders')));
  const stack = h('div', { class: 'fy-stack' });
  const acts = h('div', { class: 'fy-acts' },
    recBtn('undo', 'Вернуть', 'Вернуть прошлую книгу', () => recUndo(box)),
    recBtn('nope', 'Не то', 'Не интересно', () => recAct(box, 'nope')),
    recBtn('read', 'Читал', 'Уже читал', () => recAct(box, 'read')),
    recBtn('want', 'Хочу', 'Хочу прочитать', () => recAct(box, 'want')));
  box.replaceChildren(head, stack, acts);
  if (state.tab === 'foryou') { window.scrollTo({ top: 0, behavior: 'instant' }); if (typeof libScrollChrome === 'function') libScrollChrome(); }
  recSize(box); requestAnimationFrame(() => recSize(box));
  recFill(stack, true);
  if (!REC.st.coached) recCoach(box);
}
// The deck always fits between the header and the tab bar: no overlap, no scrolling.
function recSize(box) {
  const stack = $('.fy-stack', box), acts = $('.fy-acts', box);
  if (!stack || !box.isConnected) return;
  const top = stack.getBoundingClientRect().top + window.scrollY;
  const bar = $('#tabbar'), barTop = bar && !bar.hidden ? bar.getBoundingClientRect().top : innerHeight - 20;
  const avail = barTop - top - acts.offsetHeight - 34;
  stack.style.height = Math.round(Math.max(300, Math.min(600, avail))) + 'px';
}
addEventListener('resize', () => { const box = $('.fy'); if (box && $('.fy-stack', box)) recSize(box); });

function recBtn(kind, label, aria, fn) {
  const svg = { undo: '<path d="M9 8H5V4"/><path d="M5.5 8A7.5 7.5 0 1 1 5 13"/>', nope: '<path d="M6.5 6.5l11 11M17.5 6.5l-11 11"/>', read: '<path d="M5 12.5l4.5 4.5L19 7.5"/>', want: '<path d="M7 4h10v16l-5-3.6L7 20Z"/>' }[kind];
  const b = h('button', { class: 'fy-btn ' + kind, 'aria-label': aria, onclick: fn });
  b.innerHTML = `<svg viewBox="0 0 24 24" aria-hidden="true">${svg}</svg>`;
  return h('div', { class: 'fy-act' }, b, h('span', null, label));
}
